async function loadKitchen(){ try{ const os=await api("/orders"); const open=os.filter(x=>["open","preparing","ready"].includes(x.status)); $("kitchenOrders").innerHTML=open.map(o=>`

Mesa ${tableName(o.table_id)}

${esc(o.status)}

Mesero: ${esc(o.waiter_name||"No asignado")}

${esc(o.customer_name||"Sin cliente")}

${o.estimated_time ? `

⏳ Estimado: ${o.estimated_time} min

` : ''}
`).join("")||"

No hay pedidos pendientes.

"; }catch(e){toast(e.message)} } async function handleKitchenStatus(id, newStatus) { if (newStatus === 'preparing') { // Si cambia a preparando, le preguntamos el tiempo const mins = prompt("¿En cuántos minutos aprox. estará listo? (Ingresa solo el número o deja vacío)"); const bodyData = { status: "preparing" }; if (mins !== null && mins.trim() !== "") { bodyData.estimated_time = Number(mins) || 0; } try { await api("/orders/"+id, { method: "PATCH", body: JSON.stringify(bodyData) }); loadKitchen(); pollNotifications(); toast("Pedido en preparación."); } catch(e) { toast(e.message) } } else { // Si lo cambia a Nuevo o Listo, simplemente lo actualizamos try { await api("/orders/"+id, { method: "PATCH", body: JSON.stringify({status: newStatus}) }); loadKitchen(); pollNotifications(); if (newStatus === 'ready') toast("¡Avisando al mesero que está listo!"); } catch(e) { toast(e.message) } } }
